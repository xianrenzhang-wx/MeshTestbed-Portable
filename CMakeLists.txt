cmake_minimum_required(VERSION 3.20)
project(MeshTestbedPortable VERSION 2.1.0 LANGUAGES CXX)
add_library(mesh_core SHARED src/mesh_core.cpp src/mesh_geometry.cpp src/mesh_bezier.cpp src/mesh_analysis.cpp)
target_include_directories(mesh_core PUBLIC include)
target_compile_features(mesh_core PRIVATE cxx_std_17)
set_target_properties(mesh_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(WIN32)
  set_target_properties(mesh_core PROPERTIES PREFIX "")
endif()
if(MSVC)
  set_property(TARGET mesh_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(mesh_core PRIVATE /W4 /EHsc)
else()
  target_compile_options(mesh_core PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(mesh_core PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
install(TARGETS mesh_core RUNTIME DESTINATION mesh_testbed LIBRARY DESTINATION mesh_testbed)
install(DIRECTORY python/mesh_testbed/ DESTINATION mesh_testbed FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
install(FILES examples/demo.py examples/manufacturing_demo.py examples/analysis_demo.py DESTINATION .)
install(FILES README.md GEOMETRY.md API.md USAGE.md CHANGELOG.md DESTINATION .)
install(FILES VALIDATION.md DESTINATION . OPTIONAL)
install(FILES geometry_validation.json DESTINATION . OPTIONAL)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY tests/ DESTINATION tests PATTERN "__pycache__" EXCLUDE)
install(DIRECTORY third_party_notices/ DESTINATION third_party_notices)
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME python_api COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -v)
  set_tests_properties(python_api PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/python;MESH_CORE_LIBRARY=$<TARGET_FILE:mesh_core>")
endif()
